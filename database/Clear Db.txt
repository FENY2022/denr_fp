
USE `denr_fp_oms`;

DELETE FROM `fpoms_deficiency_notices`;

-- Step 4 evaluations
DELETE FROM `permit_step4_requirement_reviews`;
DELETE FROM `permit_step4_reviews`;

-- Step 3, payment and records
DELETE FROM `permit_payment_verifications`;
DELETE FROM `permit_records_archives`;
DELETE FROM `permit_step3_movements`;

-- Permit workflow
DELETE FROM `permit_flow_events`;
DELETE FROM `permit_flow_cases`;
DELETE FROM `permit_flow_artifacts`;

-- Step 2 application history and documents
DELETE FROM `permit_step2_events`;
DELETE FROM `permit_step2_documents`;

-- Area photo documentation
DELETE FROM `permit_step2_area_photos`;
DELETE FROM `permit_step2_area_photo_groups`;
DELETE FROM `permit_step2_area_photo_drafts`;
DELETE FROM `permit_step2_area_photo_group_drafts`;

-- Unsubmitted application drafts

-- Legacy permit data
DELETE FROM `application_documents`;
DELETE FROM `workflow_actions`;
DELETE FROM `permit_records`;

-- Parent application tables
DELETE FROM `permit_applications`;
DELETE FROM `permit_step2_applications`;

-- Clear permit-related notifications
DELETE FROM `notifications`
WHERE `route_page` LIKE 'permit%'
   OR `route_page` IN (
       'payment_notice_queue',
       'lpdd_fus_payment_queue',
       'deficiency_notice_queue',
       'records_archive'
   )
   OR `title` LIKE '%FP Application%'
   OR `title` LIKE '%FP Payment%'
   OR `title` LIKE 'Step 2%'
   OR `title` LIKE '%Template No. 2%'
   OR `title` LIKE 'Official Notice to Pay%';

-- Clear permit-related audit history only
DELETE FROM `audit_logs`
WHERE `action` LIKE 'permit.%'
   OR `target_type` IN (
       'permit_step2_application',
       'permit_application'
   );

COMMIT;

-- Verify cleanup
